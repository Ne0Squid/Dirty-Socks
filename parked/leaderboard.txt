function weeklyGains(list){
  const days=lvlHistory&&lvlHistory.days?Object.keys(lvlHistory.days).sort():[];
  if(days.length<2)return null;
  const today=days[days.length-1];
  const cut=new Date(today+"T00:00:00Z");cut.setUTCDate(cut.getUTCDate()-7);
  const cutKey=cut.toISOString().slice(0,10);
  const rows=[];let since=null;
  list.forEach(m=>{
    const s=slugOf(m);
    const have=days.filter(k=>k<today&&lvlHistory.days[k][s]!=null);
    if(!have.length)return;
    const older=have.filter(k=>k<=cutKey);
    const base=older.length?older[older.length-1]:have[0];
    if(!since||base<since)since=base;
    rows.push({m,val:m.level-lvlHistory.days[base][s]});
  });
  return {rows,since};
}
function drawBoard(){
  if(!sorted.length)return;
  const SHOW_LEVEL_TAB=false; // temporarily hidden; set to true to bring back the Highest level tab
  const modes=[];
  if(SHOW_LEVEL_TAB)modes.unshift(["level","Highest level"]);
  if(sorted.some(m=>m.ilvl))modes.push(["ilvl","Item level"]);
  modes.push(["gain","Gained this week"]);
  if(!modes.some(x=>x[0]===lbMode))lbMode=modes[0][0];
  document.getElementById("lbcard").hidden=false;
  document.getElementById("lbtabs").innerHTML=modes.map(([k,l])=>`<button class="${k===lbMode?"on":""}" data-b="${k}">${l}</button>`).join("");
  let rows=[],note="",fmtv=v=>v;
  if(lbMode==="level"){rows=sorted.map(m=>({m,val:m.level}));fmtv=v=>"Level "+v}
  else if(lbMode==="ilvl"){rows=sorted.filter(m=>m.ilvl).map(m=>({m,val:m.ilvl}));fmtv=v=>"Item level "+v}
  else{
    const g=weeklyGains(sorted);
    if(!g)note="Weekly gains appear once the site has saved a couple of days of levels. Check back tomorrow.";
    else{rows=g.rows.filter(r=>r.val>0);fmtv=v=>"+"+v+(v===1?" level":" levels");note=(rows.length?"":"No level-ups yet. ")+"Levels gained since "+fmtDayUTC(g.since)+".";}
  }
  rows.sort((a,b)=>b.val-a.val||b.m.level-a.m.level);rows=rows.slice(0,5);
  document.getElementById("lbbody").innerHTML=rows.map((r,i)=>`<div class="lbrow"><span class="lbpos">${i+1}</span>${cico(r.m)}<span><strong>${nameLink(r.m)}</strong><small>${esc(r.m.spec?r.m.spec+" ":"")}${esc(r.m.cls)}</small></span><span class="lbval">${fmtv(r.val)}</span></div>`).join("");
  document.getElementById("lbnote").textContent=note;
}
document.getElementById("lbtabs").addEventListener("click",e=>{const b=e.target.closest("button");if(b){lbMode=b.dataset.b;drawBoard()}});
